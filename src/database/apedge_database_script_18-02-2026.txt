-- ap_tenants Table
CREATE TABLE ap_tenants (
  tenant_id SERIAL PRIMARY KEY,
  tenant_name TEXT NOT NULL,
  logo_path TEXT,
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  created_by INTEGER,
  updated_by INTEGER
);

-- ap_users Table
CREATE TABLE ap_users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(100) UNIQUE NOT NULL,
  contact VARCHAR(15) NOT NULL,
  user_password TEXT NOT NULL,
  user_role VARCHAR(20),
  tenant_id INTEGER REFERENCES ap_tenants(tenant_id) ON DELETE SET NULL,
  user_name VARCHAR(150),
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  is_active BOOLEAN DEFAULT TRUE,
  metadata TEXT,
  failed_login_attempts INTEGER DEFAULT 0,
  last_failed_login TIMESTAMPTZ,
  account_locked_until TIMESTAMPTZ
);

-- ap_documents Table
CREATE TABLE ap_documents (
  doc_id SERIAL PRIMARY KEY,
  tenant_id INTEGER REFERENCES ap_tenants(tenant_id) ON DELETE SET NULL,
  user_id INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
  file_path TEXT NOT NULL,
  status TEXT,
  stage TEXT,
  doc_type TEXT,
  input_source TEXT,
  docedge_id TEXT,
  contents JSONB,
  tokens BIGINT,
  created_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  error_desc JSONB,
  page_count INTEGER
);

-- ap_invoices Table
CREATE TABLE ap_invoices (
  invoice_id SERIAL PRIMARY KEY,

  po_no TEXT,
  invoice_date DATE,
  vendor_name TEXT,
  extracted_data JSONB NOT NULL,
  status TEXT,
  error_desc JSONB,
  comments TEXT,
  stage TEXT,
  created_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  field_structure JSONB,
  extracted_data_history JSONB
);

CREATE TABLE ap_process_workflow_instances (
    instance_id SERIAL PRIMARY KEY,
    tenant_id INTEGER REFERENCES ap_tenants(tenant_id),
    module_name VARCHAR(100) NOT NULL,
    Process_name VARCHAR(100) NOT NULL,
 
    -- Current State
    current_step VARCHAR(100) NOT NULL,
    next_step VARCHAR(100) NOT NULL,
    status VARCHAR(100) NOT NULL, -- e.g., 'In Progress', 'Approved'

	  error_desc JSONB,
	
    -- Ownership & Traceability
    pending_with VARCHAR(50),
    created_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE ap_process_workflow_history (
    history_id SERIAL PRIMARY KEY,
    tenant_id INTEGER REFERENCES ap_tenants(tenant_id),
    instance_id INTEGER REFERENCES ap_process_workflow_instances(instance_id),
    module_name VARCHAR(100) NOT NULL,
    Process_name VARCHAR(100) NOT NULL,
 
    -- 1. TIME Traceability
    transaction_timestamp TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
   
    -- 2. OWNERSHIP Traceability (Who did it?)
    created_by INTEGER REFERENCES ap_users(id) NOT NULL,
    created_role_name VARCHAR(50) NOT NULL,   -- The role they assumed (e.g., 'Manager')
   
    -- 3. ACTION Traceability (What did they do?)
    previous_step VARCHAR(50),
    next_step VARCHAR(100),
    actionRef VARCHAR(100),             -- Button ID (e.g., 'btn_reject')
    user_comments TEXT,                     -- Reasoning (e.g., "Budget too high")
    status VARCHAR(100),             -- status of workflow (e.g., 'Submitted', 'approved')
   
    -- 4. DATA Traceability (Snapshot)
    -- Stores the state of the JSONB data after the change.
    history_json JSONB
);

CREATE TABLE churn_policy (
  	churn_policy_id BIGSERIAL PRIMARY KEY, -- Unique policy record ID
	  instance_id INTEGER REFERENCES ap_process_workflow_instances(instance_id),
    tenant_id INTEGER REFERENCES ap_tenants(tenant_id) ON DELETE SET NULL,
	  new_policy_number VARCHAR(100), -- New policy number issued
  	new_policy_rcd VARCHAR(100), -- New policy record identifier
  	old_policy_number VARCHAR(100), -- Previous policy number
  	fls_name VARCHAR(100), -- Field level staff name
  	fls_code VARCHAR(50), -- Field level staff code
  	channel_hod_name VARCHAR(100), -- Head of channel name
  	attachment_type VARCHAR(100), -- Type of document attachment
  	exception_reason TEXT, -- Reason for exception handling
  	old_channel VARCHAR(100), -- Channel name
  	fls_email_excel VARCHAR(100), -- FLS responsible email
  	fls_name_excel VARCHAR(100), -- FLS responsible person name
  	policy_status VARCHAR(50), -- Current status (SUBMITTED, APPROVED, REJECTED, etc.)
  	action_comments TEXT, -- Comments from users/approvers/admin
  	approver_revert_count INTEGER, -- Number of times approver reverted
  	admin_revert_count INTEGER, -- Number of times admin reverted
  	created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP, -- Record creation timestamp
  	created_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL, -- User who created policy
  	updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP, -- Last updated timestamp
  	updated_by INTEGER REFERENCES ap_users(id) ON DELETE SET NULL -- User who last updated policy
);
